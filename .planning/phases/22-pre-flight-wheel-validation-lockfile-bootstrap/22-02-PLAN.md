    ```bash
    mkdir -p /tmp/preflight-22
    docker pull quay.io/jupyter/scipy-notebook:latest
    # Inspect the :latest manifest and find the dated tag pointing to the same digest.
    LATEST_DIGEST=$(docker inspect --format='{{index .RepoDigests 0}}' quay.io/jupyter/scipy-notebook:latest | sed 's/.*@//')
    echo "scipy-notebook latest digest: ${LATEST_DIGEST}"
    # Query quay.io tag listing for the dated tag matching this digest.
    # Use docker pull on candidate dated tags from the past 14 days; pick the most recent that
    # resolves to the same digest as :latest. Tag format is YYYY-MM-DD.
    RESOLVED_TAG=""
    for d in $(seq 0 14); do
      CANDIDATE=$(date -d "${d} days ago" +%Y-%m-%d)
      if docker pull "quay.io/jupyter/scipy-notebook:${CANDIDATE}" >/dev/null 2>&1; then
        CAND_DIGEST=$(docker inspect --format='{{index .RepoDigests 0}}' "quay.io/jupyter/scipy-notebook:${CANDIDATE}" 2>/dev/null | sed 's/.*@//' || echo "")
        if [ "${CAND_DIGEST}" = "${LATEST_DIGEST}" ]; then
          RESOLVED_TAG="${CANDIDATE}"
          break
        fi
      fi
    done
    if [ -z "${RESOLVED_TAG}" ]; then
      echo "ERROR: could not resolve a dated tag matching :latest in past 14 days" >&2
      exit 3
    fi
    echo "${RESOLVED_TAG}" > /tmp/preflight-22/scipy-notebook-tag.txt
    echo "Resolved scipy-notebook tag: ${RESOLVED_TAG}"
    BASE_IMAGE="quay.io/jupyter/scipy-notebook:${RESOLVED_TAG}"
    ```

    **Step D — Define preflight function (auto-fallback per D-22-07):**

    ```bash
    preflight_python() {
      local PYVER="$1"
      local OUTLOCK="/tmp/preflight-22/requirements-${PYVER}.txt"
      local LOG="/tmp/preflight-22/run-${PYVER}.log"
      echo "=== Pre-flight for Python ${PYVER} ===" | tee "${LOG}"

      # Compile (uv vendored from astral-sh/uv:0.11.8).
      docker run --rm \
        -v "$(pwd)":/work -w /work \
        -e UV_LINK_MODE=copy \
        --entrypoint sh \
        "${BASE_IMAGE}" -c "
          set -e
          # Install uv from the official ghcr image into a tmp dir, then run.
          curl -LsSf https://astral.sh/uv/0.11.8/install.sh | sh >/dev/null 2>&1 || true
          export PATH=/root/.local/bin:\$PATH
          uv --version
          uv pip compile requirements.in --python-version ${PYVER} --generate-hashes -o ${OUTLOCK}
        " 2>&1 | tee -a "${LOG}"
      if [ ! -s "${OUTLOCK}" ]; then
        echo "FAIL: compile produced no output for Python ${PYVER}" | tee -a "${LOG}"
        return 1
      fi
      # Verify hashes are present.
      if ! grep -q -- "--hash=sha256:" "${OUTLOCK}"; then
        echo "FAIL: lockfile missing sha256 hashes for Python ${PYVER}" | tee -a "${LOG}"
        return 2
      fi
      # Install + import smoke in throwaway container (D-22-03: --system, never sync).
      docker run --rm \
        -v "$(pwd)":/work \
        -v "${OUTLOCK}":/tmp/req.txt:ro \
        -e UV_LINK_MODE=copy \
        --entrypoint sh \
        "${BASE_IMAGE}" -c "
          set -e
          curl -LsSf https://astral.sh/uv/0.11.8/install.sh | sh >/dev/null 2>&1 || true
          export PATH=/root/.local/bin:\$PATH
          uv pip install --system -r /tmp/req.txt
          python -c 'import duckdb, papermill, shap, prophet, lightgbm, xgboost, onnxruntime, optuna, umap'
        " 2>&1 | tee -a "${LOG}"
      local RC=$?
      if [ "${RC}" -ne 0 ]; then
        echo "FAIL: install/import-smoke for Python ${PYVER} (rc=${RC})" | tee -a "${LOG}"
        return 3
      fi
      echo "PASS: Python ${PYVER} pre-flight green" | tee -a "${LOG}"
      return 0
    }
    ```
